import React, {useEffect,useRef,useState} from 'react';
import {createRoot} from 'react-dom/client';
import ParallaxCarousel, {type ParallaxCarouselRef} from './parallax-carousel';
let webgl=false;try{const gl=document.createElement("canvas").getContext("webgl2");webgl=!!gl;gl?.getExtension("WEBGL_lose_context")?.loseContext()}catch{}
const stage=document.querySelector<HTMLElement>('#after')!;
const links=Array.from(stage.querySelectorAll<HTMLAnchorElement>('.gallery-open'));
const images=links.map(a=>a.href);
const labels=links.map(a=>a.dataset.caption||a.querySelector('img')?.alt||'Φωτογραφία έργου');
class Boundary extends React.Component<{children:React.ReactNode},{failed:boolean}>{
 state={failed:false}; static getDerivedStateFromError(){return {failed:true}};
 componentDidCatch(){stage.classList.remove('parallax-ready')}
 render(){return this.state.failed?null:this.props.children}
}
function Gallery(){
 const strip=useRef<HTMLDivElement>(null),drag=useRef<{x:number;scroll:number;moved:boolean}|null>(null);
 const ref=useRef<ParallaxCarouselRef>(null),host=useRef<HTMLDivElement>(null);
 const [width,setWidth]=useState(900),[index,setIndex]=useState(0),[ready,setReady]=useState(false);
 const [reduce,setReduce]=useState(false);
 useEffect(()=>{const m=matchMedia('(prefers-reduced-motion: reduce)');const f=()=>setReduce(m.matches);f();m.addEventListener('change',f);return()=>m.removeEventListener('change',f)},[]);
 useEffect(()=>{const ro=new ResizeObserver(([e])=>setWidth(e.contentRect.width));if(host.current)ro.observe(host.current);return()=>ro.disconnect()},[]);
 const mobile=width<600,imageWidth=Math.round(width*(mobile?.88:.74)),imageHeight=Math.round(imageWidth*(mobile?.8:.66));
 const go=(i:number)=>{i=Math.max(0,Math.min(images.length-1,i));if(webgl)ref.current?.scrollToIndex(i);else strip.current?.scrollTo({left:i*(imageWidth+(mobile?16:28)),behavior:reduce?"instant":"smooth"})};
 useEffect(()=>{if(!webgl){setReady(true);stage.classList.add("parallax-ready")}},[]);
 const open=()=>links[index].click();
 return <div ref={host} className="arcus-project-carousel" role="region" aria-label="Φωτογραφίες ολοκληρωμένου έργου" tabIndex={0} onKeyDown={e=>{if(e.key==='ArrowRight'){e.preventDefault();go(index+1)}if(e.key==='ArrowLeft'){e.preventDefault();go(index-1)}}}>
  <p className="carousel-hint">Σύρε για να εξερευνήσεις τις τρεις κατοικίες.</p>
  <div className="carousel-viewport" style={{height:imageHeight+12}}>
   {webgl?<ParallaxCarousel ref={ref} images={images} imageWidth={imageWidth} imageHeight={imageHeight} gap={mobile?16:28} parallaxIntensity={reduce?0:mobile?.05:.12} uvScale={.3} lerp={reduce?1:.12} autoplaySpeed={0} loop={false} borderRadius={4} showProgress={false} onIndexChange={setIndex} onReady={()=>{setReady(true);stage.classList.add('parallax-ready')}} />:<div ref={strip} className="carousel-native" style={{gap:mobile?16:28,paddingInline:(width-imageWidth)/2}} onScroll={e=>setIndex(Math.max(0,Math.min(images.length-1,Math.round(e.currentTarget.scrollLeft/(imageWidth+(mobile?16:28))))))} onPointerDown={e=>{if(e.pointerType==="touch")return;drag.current={x:e.clientX,scroll:e.currentTarget.scrollLeft,moved:false};e.currentTarget.setPointerCapture(e.pointerId)}} onPointerMove={e=>{if(drag.current){const dx=e.clientX-drag.current.x;if(Math.abs(dx)>5)drag.current.moved=true;e.currentTarget.scrollLeft=drag.current.scroll-dx}}} onPointerUp={e=>{if(drag.current?.moved){go(Math.round(e.currentTarget.scrollLeft/(imageWidth+(mobile?16:28))))}drag.current=null}} onPointerCancel={()=>drag.current=null}>{images.map((src,i)=><img key={src} src={src} alt={labels[i]} draggable={false} width={imageWidth} height={imageHeight} style={{width:imageWidth,height:imageHeight}} loading={i<2?"eager":"lazy"}/>)}</div>}
  </div>
  {ready&&<><div className="carousel-toolbar"><div className="carousel-caption"><span className="carousel-count" aria-live="polite">{String(index+1).padStart(2,'0')} / {images.length}</span><p>{labels[index]}</p></div><div className="carousel-actions"><button type="button" onClick={()=>go(index-1)} disabled={index===0} aria-label="Προηγούμενη φωτογραφία">←</button><button type="button" onClick={()=>go(index+1)} disabled={index===images.length-1} aria-label="Επόμενη φωτογραφία">→</button><button type="button" className="carousel-expand" onClick={open}>Πλήρης προβολή</button></div></div><div className="carousel-thumbnails" aria-label="Επιλογή φωτογραφίας">{images.map((src,i)=><button type="button" key={src} onClick={()=>go(i)} aria-label={`Φωτογραφία ${i+1}: ${labels[i]}`} aria-current={i===index?'true':undefined}><img src={src.replace('.webp','-640.webp')} onError={e=>{e.currentTarget.onerror=null;e.currentTarget.src=src}} alt="" loading="lazy"/></button>)}</div></>}
 </div>
}
const mount=document.querySelector('#arcus-parallax-mount');
if(mount){mount.setAttribute("data-renderer",webgl?"react-bits-webgl":"native-fallback");createRoot(mount).render(<Boundary><Gallery/></Boundary>)}
