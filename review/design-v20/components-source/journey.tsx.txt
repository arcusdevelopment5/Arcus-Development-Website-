import React,{useState,useRef,useEffect} from 'react';
import {createRoot} from 'react-dom/client';
import SpinShutter,{type SpinShutterHandle} from './spin-shutter';
const stages=[{id:'before',name:'Πριν',title:'Εδώ ξεκίνησαν όλα.',desc:'Ο χώρος στην αρχική του κατάσταση, πριν από την ανακαίνιση.'},{id:'renders',name:'3D μελέτη',title:'Η ιδέα παίρνει μορφή.',desc:'Τρισδιάστατες απεικονίσεις της πρότασης — όχι φωτογραφίες του ολοκληρωμένου έργου.'},{id:'after',name:'Αποτέλεσμα',title:'Τρεις κατοικίες. Μια νέα αρχή.',desc:'Πραγματικές φωτογραφίες από τις τρεις ολοκληρωμένες κατοικίες.'}].map(s=>({...s,links:Array.from(document.querySelectorAll<HTMLAnchorElement>(`#${s.id} .gallery-open`))}));
const filters=[{id:'all',name:'Όλες',title:'Όλη η διαδρομή του έργου.',desc:'Από την αρχική κατάσταση, στην 3D μελέτη και στις τρεις ολοκληρωμένες κατοικίες.',links:stages.flatMap(s=>s.links)},...stages];
let gpu=false;try{const g=document.createElement('canvas').getContext('webgl2');gpu=!!g;g?.getExtension('WEBGL_lose_context')?.loseContext()}catch{}
function Journey(){
 const [stage,setStage]=useState(0),[photo,setPhoto]=useState(0),[view,setView]=useState(false),[simple,setSimple]=useState(!gpu),[all,setAll]=useState(false),[playing,setPlaying]=useState(true),[reduced,setReduced]=useState(false),[dark,setDark]=useState(document.documentElement.dataset.theme==='dark');
 const engine=useRef<SpinShutterHandle>(null),touch=useRef(0);
 useEffect(()=>{const update=()=>setDark(document.documentElement.dataset.theme==='dark');const observer=new MutationObserver(update);observer.observe(document.documentElement,{attributes:true,attributeFilter:['data-theme']});return()=>observer.disconnect()},[]);
 useEffect(()=>{const resize=()=>document.documentElement.style.setProperty('--journey-viewport',`${document.documentElement.clientWidth}px`);resize();window.addEventListener('resize',resize);return()=>window.removeEventListener('resize',resize)},[]);
 const current=filters[stage];
 const items=current.links.map((a,i)=>({src:a.href,alt:a.dataset.caption||a.querySelector('img')?.alt||`${current.name} ${i+1}`,title:a.dataset.caption||a.querySelector('img')?.alt}));
 useEffect(()=>{const m=matchMedia('(max-width: 700px) and (pointer: coarse), (prefers-reduced-motion: reduce)');const update=()=>setSimple(!gpu||m.matches);update();m.addEventListener('change',update);document.body.classList.add('journey-ready');return()=>m.removeEventListener('change',update)},[]);
 useEffect(()=>{const m=matchMedia('(prefers-reduced-motion: reduce)');const update=()=>setReduced(m.matches);update();m.addEventListener('change',update);return()=>m.removeEventListener('change',update)},[]);
 useEffect(()=>{if(!simple||!playing||view||reduced)return;const timer=setInterval(()=>{if(!document.hidden)setPhoto(p=>(p+1)%items.length)},5000);return()=>clearInterval(timer)},[simple,playing,view,reduced,stage,items.length]);
 useEffect(()=>{document.body.classList.toggle('journey-all',all)},[all]);
 const selectStage=(i:number)=>{setStage(i);setPhoto(0);setView(false)};
 const step=(d:number)=>{setPhoto(p=>Math.max(0,Math.min(items.length-1,p+d)));setView(true)};
 return <><div className="journey-heading"><p className="journey-eyebrow">Η μεταμόρφωση του χώρου</p><h2>Από το πριν, στο σήμερα.</h2><p>Ακολούθησε την εξέλιξη της Αθανασίου Διάκου.</p></div>
 <div className="journey-tabs" role="tablist" aria-label="Στάδια έργου">{filters.map((s,i)=><button id={`journey-tab-${i}`} role="tab" aria-selected={stage===i} aria-controls="journey-panel" key={s.id} onClick={()=>selectStage(i)} onKeyDown={e=>{if(e.key==='ArrowRight'||e.key==='ArrowLeft'){e.preventDefault();const n=(i+(e.key==='ArrowRight'?1:filters.length-1))%filters.length;selectStage(n);document.getElementById(`journey-tab-${n}`)?.focus()}}}><span>{i===0?"∞":`0${i}`}</span>{s.name}<small>{s.links.length} εικόνες</small></button>)}</div>
 <div id="journey-panel" role="tabpanel" aria-labelledby={`journey-tab-${stage}`}>
 <div className="journey-window" data-renderer={simple?'native':'spin-shutter-webgl'} onKeyDown={e=>{if(e.key==='ArrowRight'){e.preventDefault();step(1)}if(e.key==='ArrowLeft'){e.preventDefault();step(-1)}if(e.key==='Escape')setView(false)}}>
 {!simple&&<SpinShutter key={stage} ref={engine} items={items} backgroundColor={dark?'#121815':'#eeeae2'} count={items.length} aspect={1.45} adaptiveLayout size={.46} radius={5} tilt={14} roll={0} distance={3.2} cluster={1} spread={1} intro={true} aperture={.7} speed={12/items.length} paused={!playing||view} pauseOnHover={false} haze={0} shadow={.12} dpr={1.5} wheel={false} captions={false} onFocusChange={(_,i)=>{if(!view)setPhoto(i)}} onSelect={(_,i)=>{setPhoto(i);setView(true)}} ariaLabel={`${current.name}: περιστροφική συλλογή φωτογραφιών με αυτόματη κίνηση`}/>}
 {(simple||view)&&<div className="journey-clean" onTouchStart={e=>touch.current=e.touches[0].clientX} onTouchEnd={e=>{const dx=e.changedTouches[0].clientX-touch.current;if(Math.abs(dx)>45)step(dx<0?1:-1)}}><img src={items[photo].src} alt={items[photo].alt} draggable={false}/>{!simple&&<button className="journey-return" onClick={()=>setView(false)}>← Επιστροφή στην περιήγηση</button>}</div>}
 {!simple&&!view&&<div className="journey-explore"><span>Ανακάλυψε τη μεταμόρφωση του χώρου</span><button onClick={()=>setView(true)}>Δες τις φωτογραφίες ↗</button></div>}
 </div>
 <div className="journey-tools"><div aria-live="polite"><span>{String(photo+1).padStart(2,'0')} / {items.length}</span><p>{simple||view?items[photo].alt:'Αυτόματη περιήγηση · Επίλεξε μια εικόνα για καθαρή προβολή'}</p></div><div className="journey-buttons"><button aria-label={playing&&!view?"Παύση περιήγησης":"Συνέχεια περιήγησης"} onClick={()=>{if(view){setView(false);setPlaying(true)}else setPlaying(!playing)}}>{playing&&!view?"Παύση ⏸":"Συνέχεια ▶"}</button><button onClick={()=>step(-1)} disabled={photo===0} aria-label="Προηγούμενη φωτογραφία">←</button><button onClick={()=>step(1)} disabled={photo===items.length-1} aria-label="Επόμενη φωτογραφία">→</button><button onClick={()=>{setView(true);current.links[photo].click()}}>Μεγέθυνση ↗</button></div></div>
 <div className="journey-thumbs" aria-label="Επιλογή φωτογραφίας">{items.map((item,i)=><button key={item.src} aria-label={`${current.name}: ${item.alt}`} aria-pressed={photo===i} onClick={()=>{setPhoto(i);setView(true)}}><img src={item.src} alt="" loading="lazy"/></button>)}</div>
 </div><div className="journey-bottom"><button className="journey-all-button" aria-expanded={all} onClick={()=>setAll(!all)}>{all?'Σύμπτυξη αναλυτικής συλλογής':'Αναλυτική συλλογή'} {all?'−':'+'}</button>{stage>0&&stage<filters.length-1?<button className="journey-next" onClick={()=>selectStage(stage+1)}>Επόμενο: {filters[stage+1].name} →</button>:<button className="journey-next" onClick={()=>selectStage(0)}>Όλες οι φωτογραφίες ↺</button>}</div></>;
}
const mount=document.getElementById('project-journey');if(mount)createRoot(mount).render(<Journey/>);
